import fs from 'node:fs';
import path from 'node:path';
import {pathToFileURL} from 'node:url';
import {loadIsolatedCellSources} from './prompt15-cell-isolation.mjs';
import {generateCountryCells,fileIdentity} from './prompt15-cell-core.mjs';
import {cellPreservationGate} from './prompt15-cell-preservation.mjs';
import {auditCountryData,physicalMetrics,geometryHash,partitionCoverage} from './prompt15-preservation-audit-core.mjs';
import {CELL_EXPERIMENT_POLICY as POLICY} from './prompt15-cell-policy.mjs';
import {makeEdge} from './prompt15-world-indices-core.mjs';
import {polygons,jsonBytes,digest,generateAtoms} from './prompt14-catalog-core.mjs';
import {territoryIdForAtom} from './prompt14-catalog-topology.mjs';
export function validateIsolatedCells(result,sources){
 const countryId=result.countryId,parents=sources.selected.features.filter(f=>f.properties.countryId===countryId).map(f=>({parentKey:`${countryId}:${f.properties.sourceFeatureId}`,adminId:f.properties.inputRole==='P1'?String(f.properties.sourceFeatureId):null,geometry:f.geometry})),cells=result.cells.map((c,i)=>({...c,ordinal:i+1})),byId=new Map(cells.map(c=>[c.id,c]));
 const lineage=cellPreservationGate(sources.selected.features,result.cells.map(c=>({type:'Feature',properties:c,geometry:c.geometry})));
 const selected={type:'FeatureCollection',features:cells.map(c=>({type:'Feature',properties:c,geometry:c.geometry}))},atoms=generateAtoms({selected,lock:{sourceVersion:sources.sourceVersion}}),identityErrors=atoms.flatMap((a,i)=>territoryIdForAtom(a.canonicalAtomKey)===cells[i].id?[]:[cells[i].id]);
 const membership=(values,key)=>{let start=0;return values.map(v=>{const ordinals=cells.filter(c=>c[key]===v).map(c=>c.ordinal),entry={...(key==='parentKey'?{parentKey:v}:{oldTerritoryId:v}),cellOrdinals:ordinals,start,count:ordinals.length};start+=ordinals.length;return entry;});};
 const administrativeMembership=membership(parents.map(p=>p.parentKey),'parentKey'),previousParents=sources.oldTerritories.get(countryId).map(o=>({oldTerritoryId:o.id,geometry:o.geometry})),previousMembership=membership(previousParents.map(p=>p.oldTerritoryId),'oldTerritoryId');
 const incidentEdges=[];for(const e of result.topology.edges){const left=byId.get(e.leftTerritoryId),right=e.rightTerritoryId?byId.get(e.rightTerritoryId):null;if(!left||e.rightTerritoryId&&!right)throw Error('Foreign/unknown topology reference in isolated result');for(let i=1;i<e.geometry.coordinates.length;i++)incidentEdges.push(makeEdge(cells,left.ordinal,right?.ordinal??0,e.geometry.coordinates[i-1],e.geometry.coordinates[i],incidentEdges.length+1));}
 const adjacency=new Map(cells.map(c=>[c.ordinal,(result.topology.adjacency[c.id]??[]).map(id=>{if(!byId.has(id))throw Error('Foreign/unknown adjacency');return byId.get(id).ordinal;})]));
 const expectedCells=cells.map(c=>{const measured=physicalMetrics(c.geometry);return {...c,geometryHash:geometryHash(c.geometry),areaKm2:measured.areaKm2,diameterKm:measured.diameterKm};}),tinySourceComponents=new Set();
 for(const p of parents)for(const [i,coordinates] of polygons(p.geometry).entries()){const m=physicalMetrics({type:'Polygon',coordinates});if(m.areaKm2<=POLICY.tinySourceComponent.maximumAreaKm2&&m.diameterKm<=POLICY.tinySourceComponent.maximumDiameterKm)tinySourceComponents.add(`${p.parentKey}:${i}`);}
 const audit=auditCountryData({countryId,cells,parents,previousParents,envelope:sources.p0.features.find(f=>f.properties.countryId===countryId).geometry,expectedCells,sourceIdentities:parents.filter(p=>p.adminId!==null).map(p=>p.adminId),administrativeMembership,previousMembership,incidentEdges,cellMeta:ordinal=>cells[ordinal-1],adjacency,tinySourceComponents});
 return {scope:'isolated-country-regeneration-proof',worldAdmission:false,existingCandidateAudit:false,status:audit.status==='PASS'&&lineage.pass&&!identityErrors.length?'PASS':'FAIL',lineage,identityErrors,audit,namespace:'Fresh per-country local ordinals and freshly constructed membership/edge records only. No existing 15-20/21 ordinal, edge or membership data used.',limitations:['Cross-country topology is not approved by this isolated proof. Frozen national boundary context is read-only; neighboring countries and downstream world artifacts need separate validation.']};
}
export function runIsolatedCellRegeneration({country,out,parentKey,model}){
 if(!/^[A-Z]{3}$/.test(country??''))throw Error('One explicit country required');
 if(model!==undefined&&model!=='source-doubles')throw Error('Unsupported explicit numeric model');
 const absolute=path.resolve(out??''),allowed=path.resolve('reports/prompt15/15-22-component-repair')+path.sep;if(!absolute.startsWith(allowed)||fs.existsSync(absolute))throw Error('New output directory below component-repair required');fs.mkdirSync(absolute,{recursive:true});
 const write=(name,value)=>fs.writeFileSync(path.join(absolute,name),JSON.stringify(value,null,2),{flag:'wx'}),codeFiles=['scripts/prompt15-cell-core.mjs','scripts/prompt15-cell-preservation.mjs','scripts/prompt15-cell-isolation.mjs','scripts/prompt15-isolated-cell-regeneration.mjs','scripts/prompt15-cell-policy.mjs','scripts/prompt14-catalog-core.mjs','scripts/prompt14-catalog-topology.mjs','scripts/prompt15-preservation-audit-core.mjs','scripts/prompt15-world-indices-core.mjs','package-lock.json'],codePins=codeFiles.map(fileIdentity),started=new Date().toISOString();
 let sources,result,fatalError,validation,lastCollection;const stages=[];
 try{sources=loadIsolatedCellSources([country]);if(parentKey){sources.selected.features=sources.selected.features.filter(f=>`${country}:${f.properties.sourceFeatureId}`===parentKey);if(sources.selected.features.length!==1)throw Error('Unknown failed parent');sources.p0={type:'FeatureCollection',features:[{...sources.selected.features[0],properties:{countryId:country}}]};}
  write('expected-parents.geojson',sources.selected);
  result=generateCountryCells(sources,country,sources.pins.candidate,console.log,{...(model==='source-doubles'?{nodingAttempt:true,nodingScale:null}:{}),onStage:({stage,collection,evidence})=>{if(collection.features.length)lastCollection=structuredClone(collection);const prefix=`${String(stages.length).padStart(2,'0')}-${stage}`;write(`${prefix}.geojson`,collection);write(`${prefix}.json`,evidence);stages.push({stage,geometry:`${prefix}.geojson`,evidence:`${prefix}.json`,geometrySha256:digest(jsonBytes(collection))});}});write('generation.json',result);
  // Reload expected parents from fixed source artifacts, independently of
  // generation PASS flags, coverage and repair ledgers.
  const reference=loadIsolatedCellSources([country]);if(parentKey){reference.selected.features=reference.selected.features.filter(f=>`${country}:${f.properties.sourceFeatureId}`===parentKey);validation={scope:'failed-parent-only',status:cellPreservationGate(reference.selected.features,result.cells.map(c=>({type:'Feature',properties:c,geometry:c.geometry}))).pass?'PASS':'FAIL',coverage:partitionCoverage(reference.selected.features[0].geometry,result.cells.map(c=>c.geometry)),notCountryApproval:true};}else validation=validateIsolatedCells(result,reference);write('independent-validation.json',validation);sources.referenceInputs=[...reference.inputs.used.values()];
 }catch(error){fatalError={name:error.name,message:error.message,stack:error.stack,evidence:error.evidence??null};write('failure.json',fatalError);if(lastCollection&&sources){try{write('unapproved-cells.geojson',lastCollection);const reference=loadIsolatedCellSources([country]);if(parentKey)reference.selected.features=reference.selected.features.filter(f=>`${country}:${f.properties.sourceFeatureId}`===parentKey);validation={scope:'partial-generated-stage-only',notCountryApproval:true,stage:stages.filter(s=>s.stage!=='rejected-precision-candidate').at(-1)?.stage,...cellPreservationGate(reference.selected.features,lastCollection.features,new Map(reference.oldTerritories.get(country).map(t=>[t.id,t.geometry])))};write('partial-validation.json',validation);sources.referenceInputs=[...reference.inputs.used.values()];}catch(e){write('partial-validation-error.json',{message:e.message,stack:e.stack});}}}
 const inputChanges=sources?.inputs.verifyStable()??[],codeChanges=codePins.filter(p=>{const now=fileIdentity(p.path);return p.sha256!==now.sha256||p.byteLength!==now.byteLength;}),status=!fatalError&&validation?.status==='PASS'&&!inputChanges.length&&!codeChanges.length?'PASS':'FAIL';
 const summary={scope:parentKey?'failed-parent-regeneration-diagnostic':'isolated-country-regeneration-proof',country,parentKey:parentKey??null,numericModel:model==='source-doubles'?{scale:null,automaticPrecisionRetries:false}:null,status,started,finished:new Date().toISOString(),peakRssBytes:process.resourceUsage().maxRSS*1024,versions:{node:process.version,jsts:JSON.parse(fs.readFileSync('node_modules/jsts/package.json')).version,polygonClipping:JSON.parse(fs.readFileSync('node_modules/polygon-clipping/package.json')).version},codePins,codeRoot:digest(jsonBytes(codePins)),inputs:sources?[...sources.inputs.used.values()]:[],referenceInputs:sources?.referenceInputs??[],inputChanges,codeChanges,stages,generatedCells:result?.cells.length??null,validationStatus:validation?.status??null,fatalError,existingCandidateModified:false,worldAdmission:false,existing15_22Pass:false};write('execution.json',summary);console.log(JSON.stringify({country,status,out,fatalError:fatalError?.message??null,cells:result?.cells.length??null}));return summary;
}
if(process.argv[1]&&import.meta.url===pathToFileURL(process.argv[1]).href){try{const args=process.argv.slice(2),value=name=>args[args.indexOf(name)+1],country=value('--country'),out=value('--out'),parentKey=args.includes('--parent')?value('--parent'):undefined,model=args.includes('--model')?value('--model'):undefined;if(args.length%2||!args.includes('--country')||!args.includes('--out')||args.some((a,i)=>i%2===0&&!['--country','--out','--parent','--model'].includes(a)))throw Error('Explicit --country / --out / optional --parent / --model source-doubles only');const result=runIsolatedCellRegeneration({country,out,parentKey,model});if(result.status!=='PASS')process.exitCode=1;}catch(e){console.error(e.stack);process.exitCode=1;}}
