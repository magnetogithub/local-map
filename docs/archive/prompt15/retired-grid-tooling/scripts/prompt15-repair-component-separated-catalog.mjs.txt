// Repair a single rejected disconnected operational cell without regenerating
// the 246 unaffected countries. The source root stays immutable and is pinned
// byte-for-byte before any artifact is reused.
import fs from 'node:fs';
import path from 'node:path';
import {gzipSync} from 'node:zlib';
import {pathToFileURL} from 'node:url';
import {fileIdentity,generateCountryCells} from './prompt15-cell-core.mjs';
import {ADMINISTRATIVE_ROOT,TASK_15_19_VERSION,auditCountryResult,loadWorldCellSources} from './prompt15-world-catalog.mjs';
import {compareText,digest,hashCanonical,jsonBytes,polygons} from './prompt14-catalog-core.mjs';

const SOURCE_ROOT='data/derived/prompt15/15-19-operational-catalog-v1-683e84a4bcd2dc8a1276';
const SOURCE_INDEX={path:`${SOURCE_ROOT}/artifact-index.json`,byteLength:118815,sha256:'7bb3a4e33aba1846767687ffcbb70524aa49cf7cf7d9bede5a994fbfcdd04172'};
const SOURCE_V19={path:'reports/prompt15/15-19-v19.json',byteLength:54334,sha256:'1f28e8a295613e486000eedd5a67f28a01a6064a8cb6d41e69255cfac8d98681'};
const REPAIR_COUNTRY='GBR';
const slash=value=>String(value).replaceAll('\\','/');
const readJSON=file=>JSON.parse(fs.readFileSync(file,'utf8'));
const same=(actual,expected)=>actual.byteLength===expected.byteLength&&actual.sha256===expected.sha256;
const cellEntry=cell=>({id:cell.id,sourceCountryId:cell.sourceCountryId,oldTerritoryId:cell.oldTerritoryId,adminId:cell.adminId,parentKey:cell.parentKey,parentSourceFeatureId:cell.parentSourceFeatureId,sourceComponentHash:cell.sourceComponentHash,componentIndex:cell.componentIndex,sourceTiny:cell.sourceTiny,refined:cell.refined,areaKm2:cell.areaKm2,diameterKm:cell.diameterUpperBoundKm,bbox:cell.bbox,centroid:cell.centroid});
const cellFeature=cell=>({type:'Feature',properties:{cellId:cell.id,sourceCountryId:cell.sourceCountryId,oldTerritoryId:cell.oldTerritoryId,adminId:cell.adminId,parentKey:cell.parentKey,parentSourceFeatureId:cell.parentSourceFeatureId,componentIndex:cell.componentIndex,refined:cell.refined},geometry:cell.geometry});

function assertSource(){
 const indexIdentity=fileIdentity(SOURCE_INDEX.path),v19Identity=fileIdentity(SOURCE_V19.path);if(!same(indexIdentity,SOURCE_INDEX)||!same(v19Identity,SOURCE_V19))throw Error('Component repair prerequisite identity changed');const index=readJSON(SOURCE_INDEX.path),report=readJSON(SOURCE_V19.path);if(index.artifacts?.length!==744||report.status!=='pass'||report.root!==SOURCE_ROOT)throw Error('Component repair prerequisite is not the pinned V19 PASS');let total=0;for(const artifact of index.artifacts){if(typeof artifact.path!=='string'||artifact.path.includes('\\')||path.posix.isAbsolute(artifact.path)||artifact.path.split('/').some(part=>!part||part==='.'||part==='..'))throw Error(`Unsafe source artifact path: ${artifact.path}`);const actual=fileIdentity(path.join(SOURCE_ROOT,...artifact.path.split('/')));if(!same(actual,artifact))throw Error(`Source artifact changed: ${artifact.path}`);total+=actual.byteLength;}if(total!==index.totalByteLength)throw Error('Source artifact total changed');return {index,indexIdentity,v19Identity};
}

export function repairComponentSeparatedCatalog({outputBase='data/derived/prompt15',onProgress=console.log}={}){
 const pinned=assertSource(),sources=loadWorldCellSources(),candidate=sources.pins.candidate,repairGenerator=fileIdentity('scripts/prompt15-repair-component-separated-catalog.mjs'),worldGenerator=fileIdentity('scripts/prompt15-world-catalog.mjs'),cellCore=fileIdentity('scripts/prompt15-cell-core.mjs');onProgress(`[component repair] regenerating ${REPAIR_COUNTRY}`);const result=generateCountryCells(sources,REPAIR_COUNTRY,candidate,onProgress),audit=auditCountryResult(result);if(audit.status!=='pass'||result.componentSeparation?.length!==1||result.cells.some(cell=>polygons(cell.geometry).length!==1))throw Error(`Component-separated ${REPAIR_COUNTRY} regeneration failed`);
 const repairIdentity={countryId:REPAIR_COUNTRY,cellRoot:digest(jsonBytes(result.cells.map(cell=>cell.id).sort(compareText))),componentSeparation:result.componentSeparation},inputs=[pinned.indexIdentity,pinned.v19Identity,repairGenerator,worldGenerator,cellCore,fileIdentity('scripts/prompt15-cell-policy.mjs'),fileIdentity('package-lock.json'),fileIdentity(`${ADMINISTRATIVE_ROOT}/artifact-index.json`)],runtime={node:process.version,v8:process.versions.v8,zlib:process.versions.zlib,platform:process.platform,arch:process.arch},rootKey=hashCanonical('prompt15-task-15-19-component-separated-repair.v1',{version:TASK_15_19_VERSION,inputs,runtime,repairIdentity}).slice(0,20),root=path.join(outputBase,`15-19-operational-catalog-v2-${rootKey}`),staging=path.join(outputBase,`.15-19-component-repair-${rootKey}-${process.pid}`);if(fs.existsSync(root)||fs.existsSync(staging))throw Error(`Refusing to overwrite component repair output: ${fs.existsSync(root)?root:staging}`);fs.mkdirSync(staging,{recursive:true});
 const artifacts=[],record=(relative,target)=>{const identity=fileIdentity(target);artifacts.push({path:slash(relative),sha256:identity.sha256,byteLength:identity.byteLength,compression:relative.endsWith('.gz')?'gzip':null});return identity;},write=(relative,value,compressed=false)=>{const target=path.join(staging,...relative.split('/'));fs.mkdirSync(path.dirname(target),{recursive:true});fs.writeFileSync(target,compressed?gzipSync(jsonBytes(value),{level:9,mtime:0}):jsonBytes(value),{flag:'wx'});return record(relative,target);};
 for(const artifact of pinned.index.artifacts){if(!artifact.path.startsWith(`countries/${REPAIR_COUNTRY}/`)&&!['catalog-root.json','provenance.json','summary.json'].includes(artifact.path)){const target=path.join(staging,...artifact.path.split('/'));fs.mkdirSync(path.dirname(target),{recursive:true});fs.copyFileSync(path.join(SOURCE_ROOT,...artifact.path.split('/')),target,fs.constants.COPYFILE_EXCL);record(artifact.path,target);}}
 const prefix=`countries/${REPAIR_COUNTRY}`,catalogIdentity=write(`${prefix}/catalog.json.gz`,{schemaVersion:TASK_15_19_VERSION,policyVersion:result.policyVersion,candidateId:candidate.id,countryId:REPAIR_COUNTRY,entries:result.cells.map(cellEntry)},true),geometryIdentity=write(`${prefix}/geometry.geojson.gz`,{type:'FeatureCollection',features:result.cells.map(cellFeature)},true),generation={countryId:REPAIR_COUNTRY,status:audit.status,checks:audit.checks,metrics:result.metrics,parents:result.parents,sourceComponents:result.sourceComponents,coverage:result.coverage,countryCoverage:result.countryCoverage,validity:result.validity,numericalNoding:result.numericalNoding,numericalProtectedFragmentRepairs:result.numericalProtectedFragmentRepairs,numericalCellRepairs:result.numericalCellRepairs,numericalGapRepairs:result.numericalGapRepairs,componentSeparation:result.componentSeparation,cellRoot:repairIdentity.cellRoot},generationIdentity=write(`${prefix}/generation.json.gz`,generation,true);
 const oldSummary=readJSON(`${SOURCE_ROOT}/summary.json`),oldCatalogRoot=readJSON(`${SOURCE_ROOT}/catalog-root.json`),countries=oldSummary.countries.map(country=>country.countryId===REPAIR_COUNTRY?{countryId:REPAIR_COUNTRY,status:audit.status,checks:audit.checks,metrics:result.metrics,cellRoot:generation.cellRoot,files:[{path:`${prefix}/catalog.json.gz`,sha256:catalogIdentity.sha256,byteLength:catalogIdentity.byteLength,compression:'gzip'},{path:`${prefix}/geometry.geojson.gz`,sha256:geometryIdentity.sha256,byteLength:geometryIdentity.byteLength,compression:'gzip'},{path:`${prefix}/generation.json.gz`,sha256:generationIdentity.sha256,byteLength:generationIdentity.byteLength,compression:'gzip'}]}:country),counts={...oldSummary.counts,cells:oldSummary.counts.cells-oldSummary.countries.find(country=>country.countryId===REPAIR_COUNTRY).metrics.cellCount+result.metrics.cellCount,vertices:oldSummary.counts.vertices-oldSummary.countries.find(country=>country.countryId===REPAIR_COUNTRY).metrics.vertexCount+result.metrics.vertexCount,edges:oldSummary.counts.edges-oldSummary.countries.find(country=>country.countryId===REPAIR_COUNTRY).metrics.edgeCount+result.metrics.edgeCount},checks={...oldSummary.checks,countriesPassed:countries.every(country=>country.status==='pass'),uniqueCellIds:true,componentSeparated:true},status=Object.values(checks).every(Boolean)?'pass':'fail';
 const provenance={task:'15-19',generatorVersion:TASK_15_19_VERSION,generator:repairGenerator,upstreamGenerator:worldGenerator,command:'node --expose-gc scripts/prompt15-repair-component-separated-catalog.mjs',inputs,runtime,candidate,policyVersion:result.policyVersion,administrativeInputRoot:ADMINISTRATIVE_ROOT,sourceCatalogRoot:SOURCE_ROOT,reusePolicy:'All 246 unaffected country artifacts are byte-copied only after the complete pinned source artifact tree is rehashed. GBR is independently regenerated by the current fixed component-separated policy.',runtimeRegeneration:false,productionPublication:false,scope:'Offline immutable catalog repair only; topology/membership, render delivery, migration, runtime, API, browser and cutover are not executed.'},summary={...oldSummary,status,root:slash(root),checks,counts,countries,next:status==='pass'?'15-20 (not executed)':'15-19 component repair rejected'},catalogRoot={...oldCatalogRoot,countryRoots:{...oldCatalogRoot.countryRoots,[REPAIR_COUNTRY]:generation.cellRoot},cellCount:counts.cells};write('catalog-root.json',catalogRoot);write('provenance.json',provenance);write('summary.json',summary);
 const artifactIndex={schemaVersion:TASK_15_19_VERSION,root:slash(root),artifacts:artifacts.slice().sort((first,second)=>compareText(first.path,second.path)),totalByteLength:artifacts.reduce((sum,artifact)=>sum+artifact.byteLength,0),command:provenance.command};fs.writeFileSync(path.join(staging,'artifact-index.json'),jsonBytes(artifactIndex),{flag:'wx'});if(status!=='pass')throw Object.assign(Error('Component-separated catalog repair rejected'),{staging,summary});fs.renameSync(staging,root);return {root,summary,artifactIndexIdentity:fileIdentity(path.join(root,'artifact-index.json')),repairIdentity};
}

if(process.argv[1]&&import.meta.url===pathToFileURL(process.argv[1]).href)try{const result=repairComponentSeparatedCatalog();console.log(JSON.stringify({status:result.summary.status,root:result.root,counts:result.summary.counts,artifactIndex:result.artifactIndexIdentity,repair:result.repairIdentity,next:result.summary.next}));}catch(error){console.error(JSON.stringify({status:'fail',message:error.message,staging:error.staging??null}));process.exitCode=1;}
